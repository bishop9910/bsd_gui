170: int bsd_download(bsd_handle*        handle,
171:                  const bsd_request* request,
172:                  bsd_progress_fn    progress,
173:                  void*              progress_user) {
174:   if (!handle || !handle->pool) {
175:     set_error("bsd_download: invalid handle");
176:     return BSD_ERROR_INVALID_ARG;
177:   }
178:   if (!request || !request->url) {
179:     set_error("bsd_download: request.url is required");
180:     return BSD_ERROR_INVALID_ARG;
181:   }
182:   if (!valid_struct_size(request->struct_size, sizeof(bsd_request))) {
183:     set_error("bsd_download: unsupported bsd_request.struct_size");
184:     return BSD_ERROR_INVALID_ARG;
185:   }
186: 
187:   const std::string url(request->url);
188:   const std::string output =
189:     (request->output && *request->output) ? request->output : "DEFAULT";
190:   const std::string checksum =
191:     (request->checksum && *request->checksum) ? request->checksum : "NO_HASH_CHECK";
192: 
193:   bool was_cancelled = false;
194: 
195:   try {
196:     BSD_CORE::Downloader downloader(*handle->pool, handle->options, url,
197:                                     checksum, output);
198: 
199:     if (progress) {
200:       downloader.set_progress_callback(
201:         [progress, progress_user](std::uint64_t downloaded,
202:                                   std::uint64_t total,
203:                                   double bytes_per_second,
204:                                   bool finished) {
205:           try {
206:             progress(progress_user, downloaded, total, bytes_per_second,
207:                      finished ? 1 : 0);
208:           } catch (...) {
209:             // never let client code unwind through the C boundary
210:           }
211:         });

216:       handle->current = &downloader;
