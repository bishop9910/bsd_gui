71: async function pasteFromClipboard(): Promise<void> {
72:   let text = "";
73:   try {
74:     text = await navigator.clipboard.readText();
75:   } catch {
76:     try {
77:       text = await bsd.readClipboard();

79:       message.error(`无法读取剪贴板：${String(error)}`);

90: async function startDownload(): Promise<void> {
91:   const urls = parseUrls();
92:   if (urls.length === 0) {
93:     message.warning("请先粘贴下载链接");
94:     return;
95:   }
96:   const invalid = urls.find((url) => !looksLikeUrl(url));
97:   if (invalid) {
98:     message.error(`链接必须以 http:// 或 https:// 开头：${invalid}`);
99:     return;
100:   }
101:   if (urls.length > 1 && output.value.trim()) {
102:     message.warning("批量下载时不能指定统一文件名，已忽略文件名");
103:   }
104:   if (!settings.dir.trim()) {
105:     message.warning("请先选择保存目录");
106:     return;
107:   }
108: 
109:   busy.value = true;
110:   const failures: string[] = [];
111:   let started = 0;
112:   for (const url of urls) {
113:     try {
114:       const result = await bsd.start(buildJob(url, urls.length === 1));
115:       addTask(result);
116:       started += 1;
117:     } catch (error) {
118:       failures.push(`${url}\n${String(error)}`);
119:     }
120:   }
121:   busy.value = false;
122: 
123:   if (started > 0) {
124:     urlText.value = "";
125:     output.value = "";
126:     checksumValue.value = "";
127:     filter.value = "all";
128:     message.success(urls.length === 1 ? "任务已开始下载" : `已开始 ${started} 个下载任务`);
129:   }
130:   if (failures.length > 0) {
131:     notification.error({
132:       title: `${failures.length} 个任务未能开始`,
133:       meta: "请检查链接、保存目录、校验值或请求头",
134:       content: failures.join("\n\n"),
135:       duration: 12000,
136:     });
137:   }
138: }
